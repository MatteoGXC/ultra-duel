cmake_minimum_required(VERSION 3.16)
project(UltraDuelServer LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_BUILD_RPATH "$ORIGIN")
set(YGO "${CMAKE_CURRENT_SOURCE_DIR}/../Tools/YGO Classes")
file(GLOB LUA "${YGO}/lua/src/*.c")
list(FILTER LUA EXCLUDE REGEX "/(lua|luac)\\.c$")
add_library(lua STATIC ${LUA})
target_include_directories(lua PUBLIC "${YGO}/lua/src")
target_compile_definitions(lua PRIVATE LUA_USE_LINUX)
target_link_libraries(lua PUBLIC m dl)
file(GLOB CORE "${YGO}/ocgcore/*.cpp")
add_library(ocgcore SHARED ${CORE})
target_link_libraries(ocgcore PRIVATE lua)
target_include_directories(ocgcore PUBLIC "${YGO}/ocgcore")
set(IRR "${YGO}/irrlicht")
set(IRR_FILES os.cpp CAttributes.cpp CFileList.cpp CFileSystem.cpp CLimitReadFile.cpp CMemoryFile.cpp CReadFile.cpp CWriteFile.cpp CXMLReader.cpp CXMLWriter.cpp CZipReader.cpp zlib/adler32.c zlib/crc32.c zlib/inffast.c zlib/inflate.c zlib/inftrees.c zlib/zutil.c)
list(TRANSFORM IRR_FILES PREPEND "${IRR}/source/Irrlicht/")
add_library(irrlicht STATIC ${IRR_FILES})
target_include_directories(irrlicht PUBLIC "${IRR}/include" "${IRR}/source/Irrlicht" "${IRR}/source/Irrlicht/zlib")
target_compile_definitions(irrlicht PUBLIC _IRR_STATIC_LIB_ NO_IRR_COMPILE_WITH_ZIP_ENCRYPTION_ NO_IRR_COMPILE_WITH_BZIP2_ NO__IRR_COMPILE_WITH_MOUNT_ARCHIVE_LOADER_ NO__IRR_COMPILE_WITH_PAK_ARCHIVE_LOADER_ NO__IRR_COMPILE_WITH_NPK_ARCHIVE_LOADER_ NO__IRR_COMPILE_WITH_TAR_ARCHIVE_LOADER_ NO__IRR_COMPILE_WITH_WAD_ARCHIVE_LOADER_)
file(GLOB LZMA "${YGO}/ygoserver/lzma/*.c")
add_library(clzma STATIC ${LZMA})
file(GLOB VFS "${YGO}/ygoserver/spmemvfs/*.c")
add_library(cspmemvfs STATIC ${VFS})
target_include_directories(cspmemvfs PUBLIC "${YGO}/sqlite3")
add_library(sqlite3 STATIC "${YGO}/sqlite3/sqlite3.c")
target_include_directories(sqlite3 PUBLIC "${YGO}/sqlite3")
set(SERVER_FILES gframe.cpp serverapi.cpp game.cpp deck_manager.cpp data_manager.cpp replay.cpp netserver.cpp single_duel.cpp tag_duel.cpp)
list(TRANSFORM SERVER_FILES PREPEND "${YGO}/ygoserver/")
add_library(ygoserver SHARED ${SERVER_FILES})
target_include_directories(ygoserver PRIVATE "${YGO}/ygoserver/spmemvfs")
target_compile_definitions(ygoserver PRIVATE YGOPRO_SERVER_MODE SERVER_ZIP_SUPPORT SERVER_PRO2_SUPPORT SERVER_PRO3_SUPPORT SERVER_TAG_SURRENDER_CONFIRM YGOPRO_SERVER_MODE_DISABLE_CLOUD_REPLAY)
target_link_libraries(ygoserver PRIVATE ocgcore clzma cspmemvfs sqlite3 lua irrlicht event event_pthreads pthread dl)
